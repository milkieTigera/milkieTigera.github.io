## test download cnt

<script>
async function fetchDownloadCount() {
  const res = await fetch('https://api.github.com/repos/milkieTigera/mydoc_download_count1/releases');
  const data = await res.json();
  let total = 0;
  data.forEach(release => {
    release.assets.forEach(asset => {
      total += asset.download_count;
    });
  });
  document.getElementById('download-count').innerText = total;
}
fetchDownloadCount();
</script>
